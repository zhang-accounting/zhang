import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid>
	<Card title="Plain text you own" icon="document">
		Your books are text files in a folder you control: edit them in any editor and keep them in Git. Zhang reads beancount ledgers as they are. Start with [your first ledger](/getting-started/first-ledger/) or [bring a beancount ledger](/getting-started/from-beancount/).
	</Card>
	<Card title="A web UI on your files" icon="laptop">
		Overview, journals, reports, balance sheet, budgets, commodities, documents and queries. What you record in the browser is written back to your files. [Take the tour](/getting-started/first-ledger/#a-tour-of-the-web-ui).
	</Card>
	<Card title="Lots and cost basis" icon="list-format">
		Hold investments at cost and sell them with FIFO, LIFO or STRICT booking, with exact decimal amounts. See [lots and cost basis](/guides/lots-and-cost-basis/).
	</Card>
	<Card title="Budgets" icon="pen">
		Monthly envelope budgets: assign money to a budget, then see what you spent and what is left. See [budgets](/guides/budgets/).
	</Card>
	<Card title="Documents" icon="add-document">
		Attach receipts and statements to accounts and transactions, from your files or by uploading them in the web UI. See [documents](/guides/documents/).
	</Card>
	<Card title="Queries" icon="magnifier">
		A BQL-compatible query language with `SELECT`, `BALANCES` and `JOURNAL`, in the web UI and over the API, with CSV export. See [querying](/guides/querying/).
	</Card>
	<Card title="Plugins" icon="puzzle">
		Opt-in WebAssembly plugins check or transform the ledger, or serve their own pages and APIs, with only the capabilities you grant. See [plugins](/guides/plugins/) and [writing plugins](/developers/writing-plugins/).
	</Card>
	<Card title="Deploy anywhere" icon="rocket">
		A Docker image and binaries for Linux and macOS. Keep the ledger on [local disk](/deployment/data-sources/local/), [S3](/deployment/data-sources/s3/), [WebDAV](/deployment/data-sources/webdav/) or [GitHub](/deployment/data-sources/github/), and sign in with a [password or passkeys](/deployment/authentication/).
	</Card>
</CardGrid>
